OPENQASM 3.0;
include "stdgates.inc";

gate U_pi_4 a {
p(-pi/4) a;
}
gate U_pi_2 a {
p(-pi/2) a;
}
qreg q[4];
creg c[4];
h q[0];
h q[1];
h q[2];
x q[3];
ctrl @ p(pi/4) q[0], q[3];
ctrl @ p(pi/4) q[1], q[3];
ctrl @ p(pi/4) q[1], q[3];
ctrl @ p(pi/4) q[2], q[3];
ctrl @ p(pi/4) q[2], q[3];
ctrl @ p(pi/4) q[2], q[3];
ctrl @ p(pi/4) q[2], q[3];
swap q[0], q[2];
h q[0];
ctrl @ U_pi_2 q[0], q[1];
h q[1];
ctrl @ U_pi_4 q[0], q[2];
ctrl @ U_pi_2 q[1], q[2];
h q[2];
c[0] = measure q[0];
c[1] = measure q[1];
c[2] = measure q[2];
