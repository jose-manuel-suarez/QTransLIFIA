OPENQASM 3.0;
include "stdgates.inc";

gate QFT4 a,b,c,d {
h d;
cp(pi / 2) d, c;
cp(pi / 4) d, b;
cp(pi / 8) d, a;
h c;
cp(pi / 2) c, b;
cp(pi / 4) c, a;
h b;
cp(pi / 2) b, a;
h a;
swap a, d;
swap b, c;
}
gate QFT4_dg a,b,c,d {
swap a, d;
swap b, c;
h a;
cp(-pi / 2) b, a;
h b;
cp(-pi / 4) c, a;
cp(-pi / 2) c, b;
h c;
cp(-pi / 8) d, a;
cp(-pi / 4) d, b;
cp(-pi / 2) d, c;
h d;
}
qreg q[4];
creg c[4];
h q[0];
h q[1];
h q[2];
h q[3];
QFT4 q[0], q[1], q[2], q[3];
QFT4_dg q[0], q[1], q[2], q[3];
c[0] = measure q[0];
c[1] = measure q[1];
c[2] = measure q[2];
c[3] = measure q[3];
