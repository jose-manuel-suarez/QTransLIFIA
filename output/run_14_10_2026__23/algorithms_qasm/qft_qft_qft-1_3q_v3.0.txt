OPENQASM 3.0;
include "stdgates.inc";

gate QFT3 a,b,c {
h c;
cp(pi / 2) c, b;
cp(pi / 4) c, a;
h b;
cp(pi / 2) b, a;
h a;
swap a, c;
}
gate QFT3_dg a,b,c {
swap a, c;
h a;
cp(-pi / 2) b, a;
h b;
cp(-pi / 4) c, a;
cp(-pi / 2) c, b;
h c;
}
qreg q[3];
creg c[3];
h q[0];
h q[1];
h q[2];
QFT3 q[0], q[1], q[2];
QFT3_dg q[0], q[1], q[2];
c[0] = measure q[0];
c[1] = measure q[1];
c[2] = measure q[2];
