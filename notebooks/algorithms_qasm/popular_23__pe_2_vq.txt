OPENQASM 3.0;
include "stdgates.inc";

gate cu1_pi_512 a {
p(pi / 512) a;
}
gate cu1_pi_256 a {
p(pi / 256) a;
}
qreg q[3];
creg c[3];
h q[0];
h q[1];
ctrl @ cu1_pi_512 q[1], q[2];
ctrl @ cu1_pi_256 q[0], q[2];
h q[0];
ctrl @ p(-pi/2) q[0], q[1];
h q[1];
c[0] = measure q[0];
c[1] = measure q[1];
c[2] = measure q[2];
